```python
"""
문제 1 (1주 + 2주). 공공 API 대기질 수집
=========================================
서울 대기질 API(측정소 40곳)에서 통합대기환경지수(khaiValue)를 모아 평균과 최고 측정소를 보고한다.

  - 한 페이지에 담기는 건수(numOfRows)가 학번마다 다르다 (응답이 여러 페이지로 나뉜다)
  - 학번마다 몇 곳의 khaiValue를 측정 실패('-')로 바꿨다 (원자료에도 '-'가 1곳 있다)

보고서의 문장: "응답 코드가 정상이고 수집이 완료되었다. 통합대기환경지수 평균은 ○○이다."

실행: python exam/q1.py --id 학번
수정본은 q1_fixed.py 로 저장해 같은 방식으로 실행한다.
"""

import json

from common import PRACTICE, key, parse_id, pick_int, pick_subset

sid = parse_id()
SRC = PRACTICE / "chapter2" / "data" / "input" / "airquality_seoul_2200.json"
ITEMS = json.loads(SRC.read_text(encoding="utf-8"))["response"]["body"]["items"]

PAGE_SIZE = pick_int(sid, "q1-page", 12, 28)
for i in pick_subset(sid, "q1-miss", len(ITEMS), pick_int(sid, "q1-nmiss", 2, 6)):
    ITEMS[i]["khaiValue"] = "-"


def fetch_page(page_no: int) -> dict:
    """API 한 페이지를 돌려준다. 실제 API와 같은 header/body 두 계층이다."""
    start = (page_no - 1) * PAGE_SIZE
    return {
        "header": {"resultCode": "00", "resultMsg": "NORMAL_CODE"},
        "body": {
            "totalCount": len(ITEMS), "pageNo": page_no, "numOfRows": PAGE_SIZE,
            "items": ITEMS[start:start + PAGE_SIZE],
        },
    }


resp = fetch_page(1)
header, body = resp["header"], resp["body"]
if header["resultCode"] != "00":
    raise SystemExit(f"API 오류: {header['resultMsg']}")

rows = list(body["items"])
total_count = body["totalCount"]
page_no = 1
while len(rows) < total_count:
    page_no += 1
    resp = fetch_page(page_no)
    page_header, page_body = resp["header"], resp["body"]
    if page_header["resultCode"] != "00":
        raise SystemExit(f"API 오류: {page_header['resultMsg']}")
    if page_body["totalCount"] != total_count:
        raise SystemExit("수집 중 전체 건수가 바뀌었습니다")
    if not page_body["items"]:
        raise SystemExit("전체 건수에 도달하기 전에 빈 페이지를 받았습니다")
    rows.extend(page_body["items"])
complete = len(rows) == total_count
if not complete:
    raise SystemExit(f"수집 건수 불일치: {len(rows)} / {total_count}")
valid_rows = [r for r in rows if r.get("khaiValue") not in (None, "", "-")]
n_missing = len(rows) - len(valid_rows)
values = [int(r["khaiValue"]) for r in valid_rows]
if not values:
    raise SystemExit("유효한 통합대기환경지수가 없습니다")
mean = sum(values) / len(values)
zero_filled_mean = sum(values) / len(rows)
top = sorted(valid_rows, key=lambda r: (-int(r["khaiValue"]), r["stationName"]))[0]

print(f"응답 코드 {header['resultCode']} ({header['resultMsg']})")
print(f"페이지 크기 {PAGE_SIZE}건, 요청 페이지 {page_no}개")
print(f"수집 {len(rows)}건 / 전체 {total_count}건 -- 수집 완료")
print(f"결측 {n_missing}건, 유효값 {len(values)}건, 유효값 합계 {sum(values)}")
print(f"결측을 0으로 채운 평균 {zero_filled_mean:.4f}, 유효값 평균 {mean:.4f}, 차이 {mean - zero_filled_mean:.4f}")
print(f"통합대기환경지수 평균 {mean:.2f}, 최고 {top['stationName']} ({top['khaiValue']})")
key("total_count", body["totalCount"])
key("n_collected", len(rows))
key("complete", int(complete))
key("n_missing_khai", n_missing)
key("mean_khai", float(mean))
key("max_khai_station", top["stationName"])
```

```python
3주차 실습과 같은 방식으로 도착 시각을 붙였다(기본 3분 지연, 일정 주기마다 72분 지연).
시험용으로 학번마다 지연 이벤트의 주기와 지연 허용 시간의 설정이 다르다.

운영팀의 보고: "두 계층 모두 정상 종료(종료 코드 0). 같은 규칙으로 집계했다."

실행: python exam/q2.py --id 학번
수정본은 q2_fixed.py 로 저장해 같은 방식으로 실행한다.
"""

import hashlib
import json
from datetime import datetime, timedelta

from common import PRACTICE, key, parse_id, pick_choice, pick_int

sid = parse_id()
INPUT_DIR = PRACTICE / "chapter3" / "data" / "input"
FILES = ["airquality_seoul_2200.json", "airquality_seoul_current.json"]
WINDOW_MIN = 60
GRACE_BATCH = 15                                   # 배치 계층의 지연 허용(분)
GRACE_SPEED_BEFORE = pick_choice(sid, "q2-grace", [5, 8, 10, 12])  # 수정 전 설정(비교용)
LATE_PERIOD = pick_int(sid, "q2-period", 4, 7)     # 정렬 순번이 이 값의 배수인 이벤트가 72분 늦게 도착


def load_events():
    rows = []
    for name in FILES:
        body = json.loads((INPUT_DIR / name).read_text(encoding="utf-8"))["response"]["body"]
        for item in body["items"]:
            if item.get("pm10Value") in (None, "-", ""):
                continue
            rows.append({
                "station": item["stationName"],
                "event_time": datetime.strptime(item["dataTime"], "%Y-%m-%d %H:%M"),
                "pm10": int(item["pm10Value"]),
            })
    rows.sort(key=lambda r: (r["event_time"], r["station"]))
    for i, row in enumerate(rows):
        delay = 72 if (i + 1) % LATE_PERIOD == 0 else 3
        row["arrival_time"] = row["event_time"] + timedelta(minutes=delay)
    return rows


def window_of(t):
    return t.replace(minute=0, second=0, microsecond=0)


def aggregate(rows, grace_min):
    """공유 집계: 창 경계, 도착 마감, 건수, 평균의 정의를 한 곳에 둔다."""
    buckets = {}
    for row in rows:
        win = window_of(row["event_time"])
        if row["arrival_time"] >= win + timedelta(minutes=WINDOW_MIN + grace_min):
            continue
        buckets.setdefault(win, []).append(row["pm10"])
    return {w.strftime("%Y-%m-%d %H:%M"): {"count": len(v), "pm10_mean": round(sum(v) / len(v), 2)}
            for w, v in sorted(buckets.items())}


def aggregate_batch(rows):
    return aggregate(rows, GRACE_BATCH)


def aggregate_speed(rows):
    return aggregate(rows, GRACE_BATCH)


def sha(obj):
    return hashlib.sha256(json.dumps(obj, sort_keys=True, ensure_ascii=False).encode("utf-8")).hexdigest()[:16]


events = load_events()
batch = aggregate_batch(events)
speed = aggregate_speed(events)
speed_before = aggregate(events, GRACE_SPEED_BEFORE)


def differences(left, right):
    windows = sorted(set(left) | set(right))
    empty = {"count": 0, "pm10_mean": 0.0}
    pairs = [(left.get(w, empty), right.get(w, empty)) for w in windows]
    return (
        sum(b != s for b, s in pairs),
        sum(b["count"] - s["count"] for b, s in pairs),
        max((abs(b["pm10_mean"] - s["pm10_mean"]) for b, s in pairs), default=0.0),
    )


n_diff_before, missing_events_before, max_mean_gap_before = differences(batch, speed_before)
n_diff_after, missing_events_after, max_mean_gap_after = differences(batch, speed)


def print_comparison(title, left, right):
    print(title)
    print(f"{'창':<18}{'배치 건수':>10}{'배치 평균':>11}{'스피드 건수':>12}{'스피드 평균':>12}")
    for win in sorted(set(left) | set(right)):
        b = left.get(win, {"count": 0, "pm10_mean": 0.0})
        s = right.get(win, {"count": 0, "pm10_mean": 0.0})
        print(f"{win:<18}{b['count']:>10}{b['pm10_mean']:>11.2f}{s['count']:>12}{s['pm10_mean']:>12.2f}")

print(f"이벤트 {len(events)}건, 창 길이 {WINDOW_MIN}분")
print(f"지연 주기 {LATE_PERIOD}, 수정 전 지연 허용 배치 {GRACE_BATCH}분 / 스피드 {GRACE_SPEED_BEFORE}분")
print(f"수정 후 지연 허용 양쪽 {GRACE_BATCH}분, 동일 집계 함수 사용")
print_comparison("[수정 전]", batch, speed_before)
print(f"어긋난 창 {n_diff_before}개, 빠진 이벤트 {missing_events_before}건, 최대 평균 차이 {max_mean_gap_before:.2f}")
print(f"수정 전 해시: 배치 {sha(batch)}, 스피드 {sha(speed_before)}, 일치 {int(sha(batch) == sha(speed_before))}")
print_comparison("[수정 후]", batch, speed)
print(f"어긋난 창 {n_diff_after}개, 빠진 이벤트 {missing_events_after}건, 최대 평균 차이 {max_mean_gap_after:.2f}")
print("두 계층 집계 종료")

key("n_diff_before", n_diff_before)
key("missing_events_before", missing_events_before)
key("max_mean_gap_before", max_mean_gap_before)
key("sha_batch", sha(batch))
key("sha_speed", sha(speed))
key("sha_equal", int(sha(batch) == sha(speed)))
```

Kappa는 집계 정의를 한 곳에 두고, 보존한 이벤트 로그를 같은 로직으로 다시 흘려보내는 **리플레이**로 과거 결과를 다시 만든다. 따라서 규칙 수정·검증의 중복을 줄이지만 원천 로그 저장 공간, 보존·접근 관리, 재처리 시간·컴퓨팅 자원이 필요하며, 긴 기간의 대규모 조인·집계는 배치 스캔보다 비쌀 수 있다. 이번 수정본은 공통 함수와 15분 설정을 사용해 양쪽 **80건**, 해시 **`e47adef45d61237d`**로 일치함을 확인했다. 이는 제공 파일의 로컬 재계산이며 실제 Kafka 로그 리플레이를 실행했다는 뜻은 아니다.

공공 감사에서 '재현 가능'이라고 할 범위는 **원천 이벤트가 실제 남아 있는 보존 기간과 시작·끝 시각 또는 파티션별 오프셋 범위**로 정해야 한다. 원본 이벤트와 입력 해시, 당시의 코드·설정·스키마·참조 데이터 버전, 실행 결과를 함께 보존해야 그때의 수치를 다시 검증할 수 있다. 이 문제에서 확인한 범위는 저장된 **2026-06-28 22:00와 23:00의 80건**이고, 파일에 없는 시점까지 재현할 수 있다고 주장하지 않는다. 요약 집계나 해시만으로는 삭제된 원천을 복원할 수 없으며, 해시 일치는 같은 결과를 확인하는 수단이지 그 규칙 자체의 타당성을 단독으로 증명하는 수단은 아니다.

```python
Kafka 소비자가 민원 이벤트 30건을 읽어 저장소에 적재하다가 도중에 멈췄고, 재시작해서 끝까지 읽었다.

  A. 처리한 뒤 커밋한다  (커밋 주기마다 지금까지 처리한 위치를 커밋)
  B. 읽은 직후 커밋한다  (한 묶음을 읽자마자 커밋하고 그 묶음을 처리)
```

**4-2** (6점) 수정 후 표: 수용 건수, 폐기 건수, LATE_OK 판정, LATE_DROP 판정, 창-구 행 수

>> 답:

**4-3** (3점) 내 지연 허용 값에서 LATE_OK와 LATE_DROP이 그렇게 판정된 이유를 watermark 값과 창 끝 시각으로 설명

>> 답:

내 지연 허용은 3분이며, 이벤트 시각 자체가 아니라 이벤트가 속한 창의 끝과 도착 시 적용 watermark를 비교한다.

| 이벤트 | 도착 파일 | 발생 시각 | 창 | 직전 최대 이벤트 시각 | 적용 watermark | 비교와 판정 |
|---|---|---|---|---|---|---|
| LATE_OK | batch2.json | 22:04 | [22:00, 22:10) | 22:09 | 22:09 − 3분 = 22:06 | 창 끝 22:10 > 22:06이므로 수용 |
| LATE_DROP | batch3.json | 21:49 | [21:40, 21:50) | 22:18 | 22:18 − 3분 = 22:15 | 창 끝 21:50 <= 22:15이므로 폐기 |

`LATE_OK`의 발생 시각 22:04가 watermark 22:06보다 이르더라도 창 끝은 22:10이므로 수용된다. 이 1건으로 22:00~22:10 강남구 집계가 3건에서 4건으로 늘었다. 원본은 batch2의 최대 시각 22:18을 미리 반영해 watermark를 22:15로 만들었기 때문에, 이 수용 대상까지 잘못 버려 총 12건만 반영했다. `LATE_DROP`은 수정 후에도 이미 닫힌 창에 속하므로 폐기 1건으로 남는다.


**4-4** (3점) 폐기가 집계 결과에 흔적을 남기지 않는 이유, 이를 감시하는 방법 하나, 지연 허용을 늘리면 치르는 대가

>> 답:

폐기된 이벤트는 집계 카운터를 늘리거나 새 창-구 행을 만들지 않는다. 내 결과에서 `LATE_DROP` 1건의 21:40~21:50 마포구 행이 없다는 사실만으로는 원래 민원이 없었던 것인지 지연 폐기된 것인지 구별할 수 없다. 따라서 7개 결과 행과 수용 13건만 보고 “14건 모두 정상 집계”라고 말하면 침묵 실패를 놓친다.

감시는 폐기 카운터를 수용 집계와 별도로 기록해 수행한다. 이번 수정본의 파일별 폐기 누적은 0 → 0 → 1이고, 입력 14건과 `수용 13 + 폐기 1 = 14`를 대사했다. 실제 Spark 운영에서는 이에 해당하는 `numRowsDroppedByWatermark`와 폐기율을 감시하고, 폐기 증가 시 지연 분포와 입력 지연을 조사한다. 이번 시험 실행은 표준 라이브러리로 제공 파일을 계산한 것으로, Spark 서비스를 새로 띄운 결과는 아니다.

지연 허용 3분을 늘리면 같은 관측 최대 시각에서 watermark가 더 과거로 이동해 늦은 이벤트를 수용할 여지가 커진다. 그 대신 창 상태를 더 오래 보관해 메모리·체크포인트 비용이 늘고, 창 확정과 확정 결과 제공도 늦어진다. 현재의 폐기 1건과 업무의 지연 허용 범위를 함께 보고 값을 정해야 한다.

```python
"""
문제 5 (6주). 일별 민원 배치 보고서의 지역별 건수
===================================================
구청이 7월 1~3일 민원을 지역별로 세어 보고한다. 자료는 6주차 실습의 실제 접수 기록 3일치(60건)이다.
학번마다 배치 마감 시각이 다르다. 각 날짜에서 접수 시각이 마감 시각 이전인 민원만 그날 배치에 들어간다.

참고 파일: practice/chapter6/data/input/lawd_cd_seoul.csv (지역 표준명 -> 표준 코드)

보고서의 문장: "민원 접수 건수를 지역별로 집계했고 오류 없이 종료되었다."

실행: python exam/q5.py --id 학번
수정본은 q5_fixed.py 로 저장해 같은 방식으로 실행한다.
"""

import csv
import json
from collections import Counter

from common import PRACTICE, key, parse_id, pick_int

sid = parse_id()
INPUT_DIR = PRACTICE / "chapter6" / "data" / "input"
DATES = ("2026-07-01", "2026-07-02", "2026-07-03")
_m = 16 * 60 + 10 * pick_int(sid, "q5-cutoff", 0, 21)          # 16:00 ~ 19:30, 10분 단위
CUTOFF = f"{_m // 60:02d}:{_m % 60:02d}"


def read_day(ds):
    path = INPUT_DIR / "complaints" / f"{ds}.jsonl"
    with open(path, encoding="utf-8") as f:
        return [json.loads(line) for line in f if line.strip()]


records = []
all_records = []
for ds in DATES:
    day_records = read_day(ds)
    all_records += day_records
    records += [r for r in day_records if r["created_at"][11:16] < CUTOFF]

# 결함 1: 같은 접수번호는 입력 순서상 처음 기록만 남긴다.
seen_ids = set()
unique_records = []
dup_removed = 0
daily = {ds: Counter() for ds in DATES}
duplicate_ids = []
for r in records:
    ds = r["created_at"][:10]
    daily[ds]["physical"] += 1
    if r["complaint_id"] in seen_ids:
        dup_removed += 1
        daily[ds]["dup_removed"] += 1
        duplicate_ids.append(r["complaint_id"])
        continue
    seen_ids.add(r["complaint_id"])
    unique_records.append(r)

# 이름 기준 묶음 수는 원본과 동일하게 마감 전 원문의 비어 있지 않은 값을 센다.
by_name = Counter(r["region_raw"] for r in records if r.get("region_raw"))
ALIASES = {"강남": "강남구", "마포": "마포구", "관악": "관악구"}
with open(INPUT_DIR / "lawd_cd_seoul.csv", encoding="utf-8", newline="") as f:
    region_to_code = {r["region_std"]: r["lawd_cd"] for r in csv.DictReader(f)}
code_to_region = {cd: region for region, cd in region_to_code.items()}
by_code = Counter()
missing_ids = []
unmapped_records = []
for r in unique_records:
    ds = r["created_at"][:10]
    # 결함 2: 공백을 제거한 뒤 약칭을 표준명으로 바꾼다.
    name = (r.get("region_raw") or "").strip()
    name = ALIASES.get(name, name)
    if not name:
        # 결함 3: 미기재와 사전에 없는 미매핑을 구분하고 원문/접수번호를 남긴다.
        missing_ids.append(r["complaint_id"])
        daily[ds]["missing"] += 1
        continue
    cd = region_to_code.get(name)
    if cd is None:
        unmapped_records.append(r)
        daily[ds]["unmapped"] += 1
        continue
    by_code[cd] += 1
    daily[ds]["mapped"] += 1

physical = len(records)
mapped = sum(by_code.values())
missing = len(missing_ids)
unmapped = len(unmapped_records)
# 결함 4: 분기별 보존식을 검산한 뒤에만 보고서를 출력한다.
preserved = physical == mapped + missing + unmapped + dup_removed
for ds, counts in daily.items():
    if counts["physical"] != sum(counts[k] for k in ("mapped", "missing", "unmapped", "dup_removed")):
        raise ValueError(f"{ds}: 총계 보존 실패")
if not preserved or len(unique_records) != mapped + missing + unmapped:
    raise ValueError("전체 총계 보존 실패")

print(f"배치 마감 {CUTOFF} (접수 시각 < 마감, 원천 +09:00), 전체 {len(all_records)}건 중 접수 기록 {physical}건")
print(f"마감 범위 밖 {len(all_records) - physical}건, 원문 이름 묶음 {len(by_name)}개")
for cd, n in sorted(by_code.items()):
    print(f"  {cd} {code_to_region[cd]} {n}건")
for ds, counts in daily.items():
    print(f"  {ds} physical={counts['physical']}, dup_removed={counts['dup_removed']},"
          f" mapped={counts['mapped']}, missing={counts['missing']}, unmapped={counts['unmapped']}")
print(f"중복 제거 접수번호: {duplicate_ids}")
print(f"미기재 접수번호: {missing_ids}")
for r in unmapped_records:
    print(f"미매핑: {r['complaint_id']} {r['region_raw']!r} {r['created_at']}")
print(f"검산: {physical} = {mapped} + {missing} + {unmapped} + {dup_removed}, 보존={preserved}")
print(f"집계 완료: {len(by_code)}개 표준 지역, 합계 {mapped}건")
key("cutoff", CUTOFF)
key("physical", physical)
key("dup_removed", dup_removed)
key("mapped", mapped)
key("missing", missing)
key("unmapped", unmapped)
key("preserved", int(preserved))
key("n_groups_by_name", len(by_name))
key("n_groups_by_code", len(by_code))
key("total_reported", sum(by_code.values()))
```
