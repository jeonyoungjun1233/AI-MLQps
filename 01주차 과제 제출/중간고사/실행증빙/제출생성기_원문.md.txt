```python
"""
문제 1 (1주 + 2주). 공공 API 대기질 수집
=========================================
서울 대기질 API(측정소 40곳)에서 통합대기환경지수(khaiValue)를 모아 평균과 최고 측정소를 보고한다.
자료는 2주차 실습의 실제 응답(airquality_seoul_2200.json)이다. 시험용으로 두 가지를 학번에 따라 바꿨다.
  - 한 페이지에 담기는 건수(numOfRows)가 학번마다 다르다 (응답이 여러 페이지로 나뉜다)
  - 학번마다 몇 곳의 khaiValue를 측정 실패('-')로 바꿨다 (원자료에도 '-'가 1곳 있다)

보고서의 문장: "응답 코드가 정상이고 수집이 완료되었다. 통합대기환경지수 평균은 ○○이다."

실행: python exam/q1.py --id 학번
수정본은 q1_fixed.py 로 저장해 같은 방식으로 실행한다.
"""

import json

from common import PRACTICE, key, parse_id, pick_int, pick_subset

sid = parse_id()
SRC = PRACTICE / "chapter2" / "data" / "input" / "airquality_seoul_2200.json"
ITEMS = json.loads(SRC.read_text(encoding="utf-8"))["response"]["body"]["items"]

PAGE_SIZE = pick_int(sid, "q1-page", 12, 28)
for i in pick_subset(sid, "q1-miss", len(ITEMS), pick_int(sid, "q1-nmiss", 2, 6)):
    ITEMS[i]["khaiValue"] = "-"


def fetch_page(page_no: int) -> dict:
    """API 한 페이지를 돌려준다. 실제 API와 같은 header/body 두 계층이다."""
    start = (page_no - 1) * PAGE_SIZE
    return {
        "header": {"resultCode": "00", "resultMsg": "NORMAL_CODE"},
        "body": {
            "totalCount": len(ITEMS), "pageNo": page_no, "numOfRows": PAGE_SIZE,
            "items": ITEMS[start:start + PAGE_SIZE],
        },
    }


resp = fetch_page(1)
header, body = resp["header"], resp["body"]
if header["resultCode"] != "00":
    raise SystemExit(f"API 오류: {header['resultMsg']}")

rows = list(body["items"])
total_count = body["totalCount"]
page_no = 1
while len(rows) < total_count:
    page_no += 1
    resp = fetch_page(page_no)
    page_header, page_body = resp["header"], resp["body"]
    if page_header["resultCode"] != "00":
        raise SystemExit(f"API 오류: {page_header['resultMsg']}")
    if page_body["totalCount"] != total_count:
        raise SystemExit("수집 중 전체 건수가 바뀌었습니다")
    if not page_body["items"]:
        raise SystemExit("전체 건수에 도달하기 전에 빈 페이지를 받았습니다")
    rows.extend(page_body["items"])
complete = len(rows) == total_count
if not complete:
    raise SystemExit(f"수집 건수 불일치: {len(rows)} / {total_count}")
valid_rows = [r for r in rows if r.get("khaiValue") not in (None, "", "-")]
n_missing = len(rows) - len(valid_rows)
values = [int(r["khaiValue"]) for r in valid_rows]
if not values:
    raise SystemExit("유효한 통합대기환경지수가 없습니다")
mean = sum(values) / len(values)
zero_filled_mean = sum(values) / len(rows)
top = sorted(valid_rows, key=lambda r: (-int(r["khaiValue"]), r["stationName"]))[0]

print(f"응답 코드 {header['resultCode']} ({header['resultMsg']})")
print(f"페이지 크기 {PAGE_SIZE}건, 요청 페이지 {page_no}개")
print(f"수집 {len(rows)}건 / 전체 {total_count}건 -- 수집 완료")
print(f"결측 {n_missing}건, 유효값 {len(values)}건, 유효값 합계 {sum(values)}")
print(f"결측을 0으로 채운 평균 {zero_filled_mean:.4f}, 유효값 평균 {mean:.4f}, 차이 {mean - zero_filled_mean:.4f}")
print(f"통합대기환경지수 평균 {mean:.2f}, 최고 {top['stationName']} ({top['khaiValue']})")
key("total_count", body["totalCount"])
key("n_collected", len(rows))
key("complete", int(complete))
key("n_missing_khai", n_missing)
key("mean_khai", float(mean))
key("max_khai_station", top["stationName"])
```

```python
"""
문제 2 (3주). 같은 집계를 두 계층에 두었다
===========================================
대기질 PM10을 한 시간 창으로 집계하는 파이프라인이 배치 계층과 스피드 계층에 같은 집계를 하나씩 두었다.
자료는 3주차 실습의 실제 응답 두 시점(22시, 23시) 80건이다. 원자료에는 도착 시각이 없으므로
3주차 실습과 같은 방식으로 도착 시각을 붙였다(기본 3분 지연, 일정 주기마다 72분 지연).
시험용으로 학번마다 지연 이벤트의 주기와 지연 허용 시간의 설정이 다르다.

운영팀의 보고: "두 계층 모두 정상 종료(종료 코드 0). 같은 규칙으로 집계했다."

실행: python exam/q2.py --id 학번
수정본은 q2_fixed.py 로 저장해 같은 방식으로 실행한다.
"""

import hashlib
import json
from datetime import datetime, timedelta

from common import PRACTICE, key, parse_id, pick_choice, pick_int

sid = parse_id()
INPUT_DIR = PRACTICE / "chapter3" / "data" / "input"
FILES = ["airquality_seoul_2200.json", "airquality_seoul_current.json"]
WINDOW_MIN = 60
GRACE_BATCH = 15                                   # 배치 계층의 지연 허용(분)
GRACE_SPEED_BEFORE = pick_choice(sid, "q2-grace", [5, 8, 10, 12])  # 수정 전 설정(비교용)
LATE_PERIOD = pick_int(sid, "q2-period", 4, 7)     # 정렬 순번이 이 값의 배수인 이벤트가 72분 늦게 도착


def load_events():
    rows = []
    for name in FILES:
        body = json.loads((INPUT_DIR / name).read_text(encoding="utf-8"))["response"]["body"]
        for item in body["items"]:
            if item.get("pm10Value") in (None, "-", ""):
                continue
            rows.append({
                "station": item["stationName"],
                "event_time": datetime.strptime(item["dataTime"], "%Y-%m-%d %H:%M"),
                "pm10": int(item["pm10Value"]),
            })
    rows.sort(key=lambda r: (r["event_time"], r["station"]))
    for i, row in enumerate(rows):
        delay = 72 if (i + 1) % LATE_PERIOD == 0 else 3
        row["arrival_time"] = row["event_time"] + timedelta(minutes=delay)
    return rows


def window_of(t):
    return t.replace(minute=0, second=0, microsecond=0)


def aggregate(rows, grace_min):
    """공유 집계: 창 경계, 도착 마감, 건수, 평균의 정의를 한 곳에 둔다."""
    buckets = {}
    for row in rows:
        win = window_of(row["event_time"])
        if row["arrival_time"] >= win + timedelta(minutes=WINDOW_MIN + grace_min):
            continue
        buckets.setdefault(win, []).append(row["pm10"])
    return {w.strftime("%Y-%m-%d %H:%M"): {"count": len(v), "pm10_mean": round(sum(v) / len(v), 2)}
            for w, v in sorted(buckets.items())}


def aggregate_batch(rows):
    return aggregate(rows, GRACE_BATCH)


def aggregate_speed(rows):
    return aggregate(rows, GRACE_BATCH)


def sha(obj):
    return hashlib.sha256(json.dumps(obj, sort_keys=True, ensure_ascii=False).encode("utf-8")).hexdigest()[:16]


events = load_events()
batch = aggregate_batch(events)
speed = aggregate_speed(events)
speed_before = aggregate(events, GRACE_SPEED_BEFORE)


def differences(left, right):
    windows = sorted(set(left) | set(right))
    empty = {"count": 0, "pm10_mean": 0.0}
    pairs = [(left.get(w, empty), right.get(w, empty)) for w in windows]
    return (
        sum(b != s for b, s in pairs),
        sum(b["count"] - s["count"] for b, s in pairs),
        max((abs(b["pm10_mean"] - s["pm10_mean"]) for b, s in pairs), default=0.0),
    )


n_diff_before, missing_events_before, max_mean_gap_before = differences(batch, speed_before)
n_diff_after, missing_events_after, max_mean_gap_after = differences(batch, speed)


def print_comparison(title, left, right):
    print(title)
    print(f"{'창':<18}{'배치 건수':>10}{'배치 평균':>11}{'스피드 건수':>12}{'스피드 평균':>12}")
    for win in sorted(set(left) | set(right)):
        b = left.get(win, {"count": 0, "pm10_mean": 0.0})
        s = right.get(win, {"count": 0, "pm10_mean": 0.0})
        print(f"{win:<18}{b['count']:>10}{b['pm10_mean']:>11.2f}{s['count']:>12}{s['pm10_mean']:>12.2f}")

print(f"이벤트 {len(events)}건, 창 길이 {WINDOW_MIN}분")
print(f"지연 주기 {LATE_PERIOD}, 수정 전 지연 허용 배치 {GRACE_BATCH}분 / 스피드 {GRACE_SPEED_BEFORE}분")
print(f"수정 후 지연 허용 양쪽 {GRACE_BATCH}분, 동일 집계 함수 사용")
print_comparison("[수정 전]", batch, speed_before)
print(f"어긋난 창 {n_diff_before}개, 빠진 이벤트 {missing_events_before}건, 최대 평균 차이 {max_mean_gap_before:.2f}")
print(f"수정 전 해시: 배치 {sha(batch)}, 스피드 {sha(speed_before)}, 일치 {int(sha(batch) == sha(speed_before))}")
print_comparison("[수정 후]", batch, speed)
print(f"어긋난 창 {n_diff_after}개, 빠진 이벤트 {missing_events_after}건, 최대 평균 차이 {max_mean_gap_after:.2f}")
print("두 계층 집계 종료")

key("n_diff_before", n_diff_before)
key("missing_events_before", missing_events_before)
key("max_mean_gap_before", max_mean_gap_before)
key("sha_batch", sha(batch))
key("sha_speed", sha(speed))
key("sha_equal", int(sha(batch) == sha(speed)))
```

```python
"""
문제 3 (4주). 소비자 장애 뒤에 적재된 행 수
============================================
Kafka 소비자가 민원 이벤트 30건을 읽어 저장소에 적재하다가 도중에 멈췄고, 재시작해서 끝까지 읽었다.
이벤트 ID는 4주차 실습에서 실제 브로커가 남긴 처리 기록(ch4_alo_duplicate_stream.jsonl)의 30개 고유 ID다.
소비자 동작은 4주차 규칙대로 구현한 시뮬레이션이다(브로커를 띄우지 않는다).
학번마다 커밋 주기와 장애가 난 시점이 다르다.

소비자가 따르는 규칙은 둘 중 하나다.
  A. 처리한 뒤 커밋한다  (커밋 주기마다 지금까지 처리한 위치를 커밋)
  B. 읽은 직후 커밋한다  (한 묶음을 읽자마자 커밋하고 그 묶음을 처리)

팀의 보고: "A 규칙으로 운영했다. 적재 행 수가 원천 건수 이상이므로 유실이 없고, 전달 보장은 exactly-once다."

실행: python exam/q3.py --id 학번
수정본은 q3_fixed.py 로 저장해 같은 방식으로 실행한다.
"""

import json

from common import PRACTICE, key, parse_id, pick_choice, pick_int

sid = parse_id()
STREAM = PRACTICE / "chapter4" / "data" / "input" / "ch4_alo_duplicate_stream.jsonl"

ids = []
for line in STREAM.read_text(encoding="utf-8").splitlines():
    event_id = json.loads(line)["event_id"]
    if event_id not in ids:
        ids.append(event_id)
N = len(ids)                                              # 원천 이벤트 수 (30)

COMMIT_EVERY = pick_choice(sid, "q3-commit", [3, 4, 5, 6, 8])
_base = pick_int(sid, "q3-crash", 14, 24) // COMMIT_EVERY * COMMIT_EVERY
CRASH_AT = _base + 2 + pick_int(sid, "q3-rest", 0, COMMIT_EVERY - 3)   # 이 개수만큼 처리한 직후 장애
if CRASH_AT > N - 2:
    CRASH_AT -= COMMIT_EVERY


def run_rule_a():
    """A. 처리한 뒤 커밋. 장애 뒤에는 마지막 커밋 위치부터 다시 읽는다. 적재한 (오프셋, ID) 행을 돌려준다."""
    rows, committed = [], 0
    for off in range(CRASH_AT):                           # 장애 전까지
        rows.append((off, ids[off]))
        if (off + 1) % COMMIT_EVERY == 0:
            committed = off + 1
    for off in range(committed, N):                       # 재시작 뒤
        rows.append((off, ids[off]))
    return rows


def run_rule_b():
    """B. 읽은 직후 커밋. 장애 뒤에는 커밋된 위치부터 읽는다. 적재한 (오프셋, ID) 행을 돌려준다."""
    rows, processed, off, crashed = [], 0, 0, False
    while off < N:
        end = min(off + COMMIT_EVERY, N)
        committed = end                                   # 묶음을 읽자마자 커밋
        for o in range(off, end):
            if not crashed and processed == CRASH_AT:
                crashed = True
                break                                     # 장애: 이 묶음의 나머지는 처리되지 않았다
            rows.append((o, ids[o]))
            processed += 1
        off = committed                                   # 재시작은 커밋된 위치부터
    return rows


rows = run_rule_a()
source_ids = set(ids)
loaded_ids = {event_id for _, event_id in rows}
lost_a = source_ids - loaded_ids
unexpected_a = loaded_ids - source_ids
duplicates = len(rows) - len(loaded_ids)
# event_id를 유일 키로 삼아 처음 적재한 행만 유지한다.
dedup = {}
duplicate_offsets = []
for off, event_id in rows:
    if event_id in dedup:
        duplicate_offsets.append(off)
    else:
        dedup[event_id] = (off, event_id)
rows_after_dedup = len(dedup)
rows_b = run_rule_b()
loaded_ids_b = {event_id for _, event_id in rows_b}
lost_b_ids = source_ids - loaded_ids_b
lost_b_offsets = [off for off, event_id in enumerate(ids) if event_id in lost_b_ids]
# 입력 처리 기록에서 ID와 실제 Kafka 위치의 일대일 대응도 확인한다.
positions_by_id = {}
for line in STREAM.read_text(encoding="utf-8").splitlines():
    record = json.loads(line)
    position = (record["partition"], record["offset"])
    positions_by_id.setdefault(record["event_id"], set()).add(position)
assert all(len(positions) == 1 for positions in positions_by_id.values())
source_position = {event_id: next(iter(positions)) for event_id, positions in positions_by_id.items()}
positions_a = {source_position[event_id] for _, event_id in rows}
assert len(set(source_position.values())) == len(source_ids)
assert set(dedup) == source_ids and not lost_a and not unexpected_a
assert rows_after_dedup == len(loaded_ids) == len(positions_a)
print(f"원천 {N}건, 커밋 주기 {COMMIT_EVERY}, 장애 시점 {CRASH_AT}건 처리 직후")
print(f"A 규칙 적재 행 수 {len(rows)}행")
print(f"ID 대사: 원천 {len(source_ids)}개, 적재 고유 {len(loaded_ids)}개, 유실 {len(lost_a)}건, 미지 ID {len(unexpected_a)}개")
print(f"A 규칙: 중복 {duplicates}행; 처리 후 커밋은 at-least-once이며 exactly-once 전달이 아님")
print(f"A 중복 재처리 위치(시뮬레이션 오프셋): {duplicate_offsets}")
print(f"event_id 중복 제거 후 {rows_after_dedup}행, 입력의 (파티션, 오프셋)으로 제거 후 {len(positions_a)}행")
print(f"B 규칙: 적재 {len(rows_b)}행, 고유 {len(loaded_ids_b)}개, 유실 {len(lost_b_ids)}건")
print(f"B 미처리 위치(시뮬레이션 오프셋): {lost_b_offsets}")
key("commit_every", COMMIT_EVERY)
key("crash_at", CRASH_AT)
key("rows_raw", len(rows))
key("duplicates", duplicates)
key("rows_after_dedup", rows_after_dedup)
key("lost_rule_b", len(lost_b_ids))
```

```python
"""
문제 4 (5주). 늦게 도착한 이벤트는 집계에 들어갔는가
=====================================================
구 단위 민원 이벤트를 시간 창으로 세는 스트리밍 집계다. 이벤트 14건이 파일 3개(batch1~3)로 나뉘어
차례로 도착한다. 자료는 5주차 실습의 실제 입력이다. 이 중 2건(LATE_OK, LATE_DROP)은 발생 시각보다
한참 늦게 도착한다. 학번마다 창 길이와 watermark 지연 허용 시간이 다르다.

적용 규칙(5주차): 파일 하나를 처리할 때 쓰는 watermark는 "그 파일 직전까지 관측한 최대 이벤트 시각 - 지연 허용"이다.
이벤트가 속한 창의 끝이 그 watermark 이하이면 폐기하고, 아니면 집계에 넣는다.

운영 보고: "이벤트 14건이 도착했고 모두 정상 집계되었다."

실행: python exam/q4.py --id 학번
수정본은 q4_fixed.py 로 저장해 같은 방식으로 실행한다.
"""

import json
from collections import defaultdict
from datetime import datetime, timedelta

from common import PRACTICE, key, parse_id, pick_choice

sid = parse_id()
STREAM_DIR = PRACTICE / "chapter5" / "data" / "input" / "stream"
FMT = "%Y-%m-%d %H:%M:%S"
WINDOW_MIN = pick_choice(sid, "q4-window", [5, 10])
DELAY_MIN = pick_choice(sid, "q4-delay", [2, 3, 4, 5, 6, 8, 12, 29, 30, 35])


def load_batches():
    batches = []
    for path in sorted(STREAM_DIR.glob("*.json")):
        rows = []
        for line in path.read_text(encoding="utf-8").splitlines():
            if line.strip():
                row = json.loads(line)
                row["t"] = datetime.strptime(row["event_time"], FMT)
                rows.append(row)
        batches.append((path.name, rows))
    return batches


def window_of(t):
    start = t.replace(minute=(t.minute // WINDOW_MIN) * WINDOW_MIN, second=0, microsecond=0)
    return start, start + timedelta(minutes=WINDOW_MIN)


batches = load_batches()
delay = timedelta(minutes=DELAY_MIN)
counts = defaultdict(int)
max_seen = None
dropped = 0
late_events = {}
received = sum(len(rows) for _, rows in batches)

print(f"창 {WINDOW_MIN}분, 지연 허용 {DELAY_MIN}분")
for name, rows in batches:
    # 이 파일에는 직전 파일까지의 최대 시각에서 구한 watermark만 적용한다.
    watermark = None if max_seen is None else max_seen - delay
    batch_accepted = batch_dropped = 0
    for r in rows:
        w_start, w_end = window_of(r["t"])
        is_dropped = watermark is not None and w_end <= watermark
        decision = "폐기" if is_dropped else "수용"
        if is_dropped:
            dropped += 1
            batch_dropped += 1
        else:
            counts[(w_start, r["district"])] += 1
            batch_accepted += 1
        if r["event_id"] in ("LATE_OK", "LATE_DROP"):
            late_events[r["event_id"]] = {
                "file": name, "event_time": r["t"],
                "window_start": w_start, "window_end": w_end,
                "watermark": watermark, "decision": decision,
            }
    batch_max = max(r["t"] for r in rows)
    max_seen = batch_max if max_seen is None else max(max_seen, batch_max)
    watermark_text = "없음" if watermark is None else watermark.strftime("%H:%M")
    print(f"{name}: 도착 {len(rows)}건, 수용 {batch_accepted}건, 폐기 {batch_dropped}건, 적용 watermark {watermark_text}, 폐기 누적 {dropped}")
    print(f"  처리 후 최대 이벤트 시각 {max_seen:%H:%M}, 다음 watermark {max_seen - delay:%H:%M}")

accepted = sum(counts.values())
assert received == accepted + dropped
print(f"집계 완료: 도착 {received}건 = 수용 {accepted}건 + 폐기 {dropped}건, 창-구 {len(counts)}행")
for event_id in ("LATE_OK", "LATE_DROP"):
    trace = late_events[event_id]
    watermark_text = "없음" if trace["watermark"] is None else trace["watermark"].strftime("%H:%M")
    print(f"{event_id}: {trace['file']}, 발생 {trace['event_time']:%H:%M}, 창 {trace['window_start']:%H:%M}-{trace['window_end']:%H:%M}, watermark {watermark_text}, {trace['decision']}")
print("창 시작 | 창 끝 | 구 | 건수")
for (start, district), count in sorted(counts.items()):
    print(f"{start:%H:%M} | {start + timedelta(minutes=WINDOW_MIN):%H:%M} | {district} | {count}")
key("window_min", WINDOW_MIN)
key("delay_min", DELAY_MIN)
key("accepted", accepted)
key("dropped", dropped)
key("n_rows", len(counts))
key("late_ok", late_events["LATE_OK"]["decision"])
key("late_drop", late_events["LATE_DROP"]["decision"])
```

```python
"""
문제 5 (6주). 일별 민원 배치 보고서의 지역별 건수
===================================================
구청이 7월 1~3일 민원을 지역별로 세어 보고한다. 자료는 6주차 실습의 실제 접수 기록 3일치(60건)이다.
학번마다 배치 마감 시각이 다르다. 각 날짜에서 접수 시각이 마감 시각 이전인 민원만 그날 배치에 들어간다.

참고 파일: practice/chapter6/data/input/lawd_cd_seoul.csv (지역 표준명 -> 표준 코드)

보고서의 문장: "민원 접수 건수를 지역별로 집계했고 오류 없이 종료되었다."

실행: python exam/q5.py --id 학번
수정본은 q5_fixed.py 로 저장해 같은 방식으로 실행한다.
"""

import csv
import json
from collections import Counter

from common import PRACTICE, key, parse_id, pick_int

sid = parse_id()
INPUT_DIR = PRACTICE / "chapter6" / "data" / "input"
DATES = ("2026-07-01", "2026-07-02", "2026-07-03")
_m = 16 * 60 + 10 * pick_int(sid, "q5-cutoff", 0, 21)          # 16:00 ~ 19:30, 10분 단위
CUTOFF = f"{_m // 60:02d}:{_m % 60:02d}"


def read_day(ds):
    path = INPUT_DIR / "complaints" / f"{ds}.jsonl"
    with open(path, encoding="utf-8") as f:
        return [json.loads(line) for line in f if line.strip()]


records = []
all_records = []
for ds in DATES:
    day_records = read_day(ds)
    all_records += day_records
    records += [r for r in day_records if r["created_at"][11:16] < CUTOFF]

# 결함 1: 같은 접수번호는 입력 순서상 처음 기록만 남긴다.
seen_ids = set()
unique_records = []
dup_removed = 0
daily = {ds: Counter() for ds in DATES}
duplicate_ids = []
for r in records:
    ds = r["created_at"][:10]
    daily[ds]["physical"] += 1
    if r["complaint_id"] in seen_ids:
        dup_removed += 1
        daily[ds]["dup_removed"] += 1
        duplicate_ids.append(r["complaint_id"])
        continue
    seen_ids.add(r["complaint_id"])
    unique_records.append(r)

# 이름 기준 묶음 수는 원본과 동일하게 마감 전 원문의 비어 있지 않은 값을 센다.
by_name = Counter(r["region_raw"] for r in records if r.get("region_raw"))
ALIASES = {"강남": "강남구", "마포": "마포구", "관악": "관악구"}
with open(INPUT_DIR / "lawd_cd_seoul.csv", encoding="utf-8", newline="") as f:
    region_to_code = {r["region_std"]: r["lawd_cd"] for r in csv.DictReader(f)}
code_to_region = {cd: region for region, cd in region_to_code.items()}
by_code = Counter()
missing_ids = []
unmapped_records = []
for r in unique_records:
    ds = r["created_at"][:10]
    # 결함 2: 공백을 제거한 뒤 약칭을 표준명으로 바꾼다.
    name = (r.get("region_raw") or "").strip()
    name = ALIASES.get(name, name)
    if not name:
        # 결함 3: 미기재와 사전에 없는 미매핑을 구분하고 원문/접수번호를 남긴다.
        missing_ids.append(r["complaint_id"])
        daily[ds]["missing"] += 1
        continue
    cd = region_to_code.get(name)
    if cd is None:
        unmapped_records.append(r)
        daily[ds]["unmapped"] += 1
        continue
    by_code[cd] += 1
    daily[ds]["mapped"] += 1

physical = len(records)
mapped = sum(by_code.values())
missing = len(missing_ids)
unmapped = len(unmapped_records)
# 결함 4: 분기별 보존식을 검산한 뒤에만 보고서를 출력한다.
preserved = physical == mapped + missing + unmapped + dup_removed
for ds, counts in daily.items():
    if counts["physical"] != sum(counts[k] for k in ("mapped", "missing", "unmapped", "dup_removed")):
        raise ValueError(f"{ds}: 총계 보존 실패")
if not preserved or len(unique_records) != mapped + missing + unmapped:
    raise ValueError("전체 총계 보존 실패")

print(f"배치 마감 {CUTOFF} (접수 시각 < 마감, 원천 +09:00), 전체 {len(all_records)}건 중 접수 기록 {physical}건")
print(f"마감 범위 밖 {len(all_records) - physical}건, 원문 이름 묶음 {len(by_name)}개")
for cd, n in sorted(by_code.items()):
    print(f"  {cd} {code_to_region[cd]} {n}건")
for ds, counts in daily.items():
    print(f"  {ds} physical={counts['physical']}, dup_removed={counts['dup_removed']},"
          f" mapped={counts['mapped']}, missing={counts['missing']}, unmapped={counts['unmapped']}")
print(f"중복 제거 접수번호: {duplicate_ids}")
print(f"미기재 접수번호: {missing_ids}")
for r in unmapped_records:
    print(f"미매핑: {r['complaint_id']} {r['region_raw']!r} {r['created_at']}")
print(f"검산: {physical} = {mapped} + {missing} + {unmapped} + {dup_removed}, 보존={preserved}")
print(f"집계 완료: {len(by_code)}개 표준 지역, 합계 {mapped}건")
key("cutoff", CUTOFF)
key("physical", physical)
key("dup_removed", dup_removed)
key("mapped", mapped)
key("missing", missing)
key("unmapped", unmapped)
key("preserved", int(preserved))
key("n_groups_by_name", len(by_name))
key("n_groups_by_code", len(by_code))
key("total_reported", sum(by_code.values()))
```

### 수정한 코드 전문 (`q6_fixed.py`)

```python
"""
문제 6 (7주). 훈련 샘플에 붙인 피처
====================================
구별 일일 민원 건수를 피처로 삼아, 인력 배치 모델의 훈련 샘플 12행에 피처를 붙인다.
피처 값은 7주차 실습이 읽는 6주차 확정 집계(3일치 x 3개 구 = 9행)이다.
피처의 타임스탬프는 "값이 알려진 시점"이며, 대상일의 집계는 그날이 끝나야 확정되므로 대상일 다음 날 00:00(UTC)이다.
피처의 유효 기간(TTL)은 3일이다. 학번마다 훈련 샘플의 구와 기준 시점(7월 1일~10일 09:00 UTC)이 다르다.

팀의 보고: "훈련 샘플 12행 전부에 피처를 붙였다. 훈련 셋이 줄지 않았으므로 조인은 정상이다."

실행: python exam/q6.py --id 학번
수정본은 q6_fixed.py 로 저장해 같은 방식으로 실행한다.
"""

import json
from datetime import datetime, timedelta, timezone

from common import PRACTICE, key, parse_id, pick_choice, pick_int

sid = parse_id()
INPUT_DIR = PRACTICE / "chapter7" / "data" / "input" / "ch6_daily"
DAYS = ["2026-07-01", "2026-07-02", "2026-07-03"]
TTL = timedelta(days=3)
CDS = ["11440", "11620", "11680"]


def build_feature_rows():
    rows = []
    for day in DAYS:
        summary = json.loads((INPUT_DIR / day / "daily_summary.json").read_text(encoding="utf-8"))
        known_at = datetime.fromisoformat(day + "T00:00:00+00:00") + timedelta(days=1)
        for r in summary["by_region"]:
            rows.append({"lawd_cd": r["lawd_cd"], "covers": day, "event_timestamp": known_at,
                         "complaint_count": int(r["count"])})
    return rows


SAMPLES = []
for i in range(12):
    cd = pick_choice(sid, f"q6-cd-{i}", CDS)
    day = pick_int(sid, f"q6-day-{i}", 0, 9)
    SAMPLES.append((cd, datetime(2026, 7, 1 + day, 9, 0, tzinfo=timezone.utc)))
SAMPLES.sort(key=lambda s: (s[1], s[0]))


def latest_value(rows, cd):
    cands = [r for r in rows if r["lawd_cd"] == cd]
    return max(cands, key=lambda r: r["event_timestamp"]) if cands else None


def latest_available(rows, cd, as_of):
    # 결함 1: 기준 시점까지 알려진 행에서만 최신값을 선택한다.
    cands = [r for r in rows if r["lawd_cd"] == cd and r["event_timestamp"] <= as_of]
    return max(cands, key=lambda r: r["event_timestamp"]) if cands else None


rows = build_feature_rows()
joined = []
n_dropped_no_value = 0
n_dropped_ttl = 0
n_leaked_future = 0
dropped = []
for cd, as_of in SAMPLES:
    original_hit = latest_value(rows, cd)
    if original_hit is not None and original_hit["event_timestamp"] > as_of:
        n_leaked_future += 1
    hit = latest_available(rows, cd, as_of)
    if hit is None:
        n_dropped_no_value += 1
        dropped.append({"lawd_cd": cd, "as_of": as_of, "reason": "값 없음", "row": None})
        continue
    # 결함 2: 알려진 값도 TTL 3일을 초과하면 훈련 샘플에 붙이지 않는다.
    if as_of - hit["event_timestamp"] > TTL:
        n_dropped_ttl += 1
        dropped.append({"lawd_cd": cd, "as_of": as_of, "reason": "TTL 초과", "row": hit})
        continue
    joined.append({"lawd_cd": cd, "as_of": as_of, "row": hit})

print(f"피처 원천 {len(rows)}행, 훈련 샘플 {len(SAMPLES)}행")
for j in joined:
    print(f"  {j['lawd_cd']}  기준 {j['as_of']:%m-%d %H:%M} UTC  건수 {j['row']['complaint_count']}"
          f"  대상일 {j['row']['covers']}  알려진 시점 {j['row']['event_timestamp'].isoformat()}"
          f"  경과 {(j['as_of'] - j['row']['event_timestamp']).total_seconds() / 3600:g}시간")
for j in dropped:
    detail = ""
    if j["row"] is not None:
        detail = (f"  최근 알려진 시점 {j['row']['event_timestamp'].isoformat()}"
                  f"  경과 {(j['as_of'] - j['row']['event_timestamp']).total_seconds() / 3600:g}시간")
    print(f"  탈락 {j['lawd_cd']}  기준 {j['as_of']:%m-%d %H:%M} UTC  사유 {j['reason']}{detail}")
assert len(SAMPLES) == len(joined) + n_dropped_no_value + n_dropped_ttl
assert all(timedelta(0) <= j["as_of"] - j["row"]["event_timestamp"] <= TTL for j in joined)
print(f"조인 결과 {len(joined)}행 -- 피처 부착 완료")
key("n_samples", len(SAMPLES))
key("n_joined", len(joined))
key("n_dropped_no_value", n_dropped_no_value)
key("n_dropped_ttl", n_dropped_ttl)
key("n_leaked_future", n_leaked_future)
key("sum_count", sum(j["row"]["complaint_count"] for j in joined))
```
